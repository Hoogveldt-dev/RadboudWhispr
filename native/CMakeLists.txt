cmake_minimum_required(VERSION 3.22)
project(collegewhispr_jni C CXX)

# Builds one self-contained shared library (whisper + ggml linked statically)
# Usage: -DVARIANT=v82 (armv8.2-a + fp16 + dotprod) or -DVARIANT=base (armv8-a)
set(WHISPER_DIR "" CACHE PATH "Path to whisper.cpp source")
set(VARIANT "base" CACHE STRING "CPU variant")

set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(VARIANT STREQUAL "v82")
    set(GGML_CPU_ARM_ARCH "armv8.2-a+fp16+dotprod" CACHE STRING "" FORCE)
    set(LIB_NAME whisper_jni_v82)
else()
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
    set(LIB_NAME whisper_jni_base)
endif()

add_subdirectory(${WHISPER_DIR} whisper EXCLUDE_FROM_ALL)

add_library(${LIB_NAME} SHARED jni.c)
if(ANDROID)
    target_link_libraries(${LIB_NAME} PRIVATE whisper log android)
else()
    # Desktop test build: stub Android logging, use the JDK's jni.h
    find_package(JNI REQUIRED)
    target_include_directories(${LIB_NAME} PRIVATE ${JNI_INCLUDE_DIRS} ${CMAKE_SOURCE_DIR}/host-stubs)
    target_link_libraries(${LIB_NAME} PRIVATE whisper)
endif()
target_compile_options(${LIB_NAME} PRIVATE -O3)
if(ANDROID)
    target_link_options(${LIB_NAME} PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
endif()
